【代码示例】

基础使用（带置信度和声纹）：

    from model import SenseVoiceSmall
    from speaker_embedding import SpeakerIdentifier

    # 加载模型
    model, kwargs = SenseVoiceSmall.from_pretrained(
        model="iic/SenseVoiceSmall",
        device="cuda:0"
    )
    
    # 创建说话人识别器
    speaker_id = SpeakerIdentifier()
    
    # 识别
    result = model.inference(
        data_in="audio.wav",
        language="auto",
        output_confidence=True,              # 新增
        output_speaker_embedding=True,       # 新增
        speaker_identifier=speaker_id,       # 新增
        **kwargs
    )
    
    # 输出结果
    print(f"文本: {result[0][0]['text']}")
    print(f"置信度: {result[0][0]['average_confidence']:.2%}")
    print(f"说话人: {result[0][0]['speaker_info']['speaker_id']}")
